DESCRIPTION
  Show current authority, documentAuthoritative state, retained document provenance, manifest formats, and recoverable revision paths for DIAGRAM_ID. This is strictly offline and does not rebuild or mutate the record.

USAGE
  sketchi show [flags] <diagram-id>

ARGUMENTS
  diagram-id string

FLAGS
  --output text|json    Result presentation format. (choices: text, json)

GLOBAL FLAGS
  --help, -h                          Show help information
  --version, -v                       Show version information
  --completions <bash|zsh|fish|sh>    Print shell completion script (choices: bash, zsh, fish, sh)

EXAMPLES
  # Read a record as a stable JSON envelope.
  sketchi show release-flow --output json
